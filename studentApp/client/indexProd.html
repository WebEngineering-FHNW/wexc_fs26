    import {SiteController} from "../../framework/client/js/controller/siteController.js";

    // app starter specific imports

    import {ARTIST}            from "./js/appTypes.js";
    import {appProjectors}     from "./js/appProjectors.js";
    import {ProdServiceFactory} from "../../framework/client/js/service/prod/prodServiceFactory.js";

    const siteController = SiteController();
    siteController.setProjectors(appProjectors);
    const baseUrl = window.location.origin;
    siteController.setServiceFactory(ProdServiceFactory(baseUrl))
        .then( _ => {
            const siteProjector  = siteController.getProjectors().siteProjector(siteController);
            document.body.append(...siteProjector.projectBodyContent());
            return siteController.findEntity(ARTIST, "0");
        })
        .then( entity => {
            siteController.setSelection({ entityName: ARTIST, entity })
        } );
